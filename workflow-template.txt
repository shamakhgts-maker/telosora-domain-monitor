name: Live domain health

on:
  schedule:
    # Offset from the busiest hour boundary. GitHub schedules are best-effort.
    - cron: '7,22,37,52 * * * *'
  workflow_dispatch:
    inputs:
      simulate:
        description: 'Send isolated simulated failure and recovery emails (no domain requests)'
        type: boolean
        default: false

permissions:
  contents: read
  issues: write

concurrency:
  group: telosora-live-domain-health
  cancel-in-progress: false

jobs:
  monitor:
    runs-on: ubuntu-latest
    timeout-minutes: 8
    env:
      GH_TOKEN: ${{ github.token }}
      GH_REPOSITORY: ${{ github.repository }}
      MONITOR_RUN_URL: ${{ github.server_url }}/${{ github.repository }}/actions/runs/${{ github.run_id }}
      SMTP_HOST: ${{ secrets.SMTP_HOST }}
      SMTP_PORT: ${{ secrets.SMTP_PORT }}
      SMTP_SECURE: ${{ secrets.SMTP_SECURE }}
      SMTP_USER: ${{ secrets.SMTP_USER }}
      SMTP_PASSWORD: ${{ secrets.SMTP_PASSWORD }}
      SMTP_FROM: ${{ secrets.SMTP_FROM }}
      ALERT_TO: info@telosora.com
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '22'
      - name: Check domains or exercise isolated notifications
        env:
          SIMULATE: ${{ github.event_name == 'workflow_dispatch' && inputs.simulate && 'true' || 'false' }}
        run: node artifacts/telosora-store/scripts/external-monitor/run-monitor.mjs
      - name: Preserve original reports even if checks or email delivery fail
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: domain-health-${{ github.run_id }}-${{ github.run_attempt }}
          path: domain-health-reports/
          if-no-files-found: error
          retention-days: 90
